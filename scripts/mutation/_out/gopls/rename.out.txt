package sample

import (
	"errors"
	"fmt"
	"strings"
	"time"
	"unicode"
)

// ValidateEmail does a minimal structural check (not full RFC 5322) — good
// enough to exercise realistic branching without pulling in regexp.
func ValidateEmail(email string) error {
	if email == "" {
		return errors.New("sample: email is empty")
	}
	if strings.ContainsAny(email, " \t\n") {
		return errors.New("sample: email contains whitespace")
	}
	at := strings.Index(email, "@")
	if at <= 0 || at == len(email)-1 {
		return errors.New("sample: email missing a valid '@'")
	}
	domain := email[at+1:]
	dot := strings.LastIndex(domain, ".")
	if dot <= 0 || dot == len(domain)-1 {
		return errors.New("sample: email domain missing a valid '.'")
	}
	return nil
}

// MaskSensitive replaces everything but the last `visible` characters of s
// with '*', the way you'd mask a card number or token in a log line.
func MaskSensitive(s string, visible int) string {
	if visible < 0 {
		visible = 0
	}
	if visible >= len(s) {
		return s
	}
	return strings.Repeat("*", len(s)-visible) + s[len(s)-visible:]
}

// DeduplicateStrings returns items with duplicates removed, preserving first-seen
// order.
func DeduplicateStrings(items []string) []string {
	seen := make(map[string]bool, len(items))
	out := make([]string, 0, len(items))
	for _, item := range items {
		if seen[item] {
			continue
		}
		seen[item] = true
		out = append(out, item)
	}
	return out
}

// ChunkSlice splits items into consecutive chunks of at most size elements.
func ChunkSlice(items []int, size int) ([][]int, error) {
	if size <= 0 {
		return nil, errors.New("sample: chunk size must be positive")
	}
	var chunks [][]int
	for i := 0; i < len(items); i += size {
		end := i + size
		if end > len(items) {
			end = len(items)
		}
		chunks = append(chunks, items[i:end])
	}
	return chunks, nil
}

// Paginate returns the 1-indexed page of items for the given page size.
func Paginate(items []string, page, pageSize int) ([]string, error) {
	if page < 1 {
		return nil, errors.New("sample: page must be >= 1")
	}
	if pageSize < 1 {
		return nil, errors.New("sample: pageSize must be >= 1")
	}
	start := (page - 1) * pageSize
	if start >= len(items) {
		return []string{}, nil
	}
	end := start + pageSize
	if end > len(items) {
		end = len(items)
	}
	return items[start:end], nil
}

// MergeConfigs merges override on top of base, without mutating either
// input map.
func MergeConfigs(base, override map[string]string) map[string]string {
	merged := make(map[string]string, len(base)+len(override))
	for k, v := range base {
		merged[k] = v
	}
	for k, v := range override {
		merged[k] = v
	}
	return merged
}

// TruncateWithEllipsis shortens s to at most maxLen bytes, appending "..."
// when truncation actually happens.
func TruncateWithEllipsis(s string, maxLen int) string {
	if maxLen <= 0 {
		return ""
	}
	if len(s) <= maxLen {
		return s
	}
	if maxLen <= 3 {
		return s[:maxLen]
	}
	return s[:maxLen-3] + "..."
}

// hasUpper, hasDigit, and hasSpecial are small helpers that
// ValidatePasswordStrength calls — deliberately factored out so there's a
// real call site for gopls's "inline function call" refactor to work on.
func hasUpper(s string) bool {
	for _, r := range s {
		if unicode.IsUpper(r) {
			return true
		}
	}
	return false
}

func hasDigit(s string) bool {
	for _, r := range s {
		if unicode.IsDigit(r) {
			return true
		}
	}
	return false
}

func hasSpecial(s string) bool {
	for _, r := range s {
		if unicode.IsPunct(r) || unicode.IsSymbol(r) {
			return true
		}
	}
	return false
}

// ValidatePasswordStrength returns a human-readable reason for every
// strength rule pw fails; an empty slice means pw passes all of them.
func ValidatePasswordStrength(pw string) []string {
	var problems []string
	if len(pw) < 8 {
		problems = append(problems, "must be at least 8 characters")
	}
	if !hasUpper(pw) {
		problems = append(problems, "must contain an uppercase letter")
	}
	if !hasDigit(pw) {
		problems = append(problems, "must contain a digit")
	}
	if !hasSpecial(pw) {
		problems = append(problems, "must contain a special character")
	}
	return problems
}

// RetryWithBackoff calls fn until it succeeds or maxAttempts is reached,
// calling sleep between attempts (injected so tests don't need real time.Sleep).
func RetryWithBackoff(fn func() error, maxAttempts int, sleep func(time.Duration)) error {
	if maxAttempts < 1 {
		return errors.New("sample: maxAttempts must be >= 1")
	}
	var lastErr error
	for attempt := 1; attempt <= maxAttempts; attempt++ {
		lastErr = fn()
		if lastErr == nil {
			return nil
		}
		if attempt < maxAttempts {
			sleep(time.Duration(attempt) * time.Millisecond)
		}
	}
	return fmt.Errorf("sample: all %d attempts failed: %w", maxAttempts, lastErr)
}

// CircuitBreakerAllow reports whether a call should be allowed through,
// given how many consecutive failures have happened, the failure
// threshold, when the last failure was, and how long the breaker stays
// open (cooldown) once tripped.
func CircuitBreakerAllow(consecutiveFailures, threshold int, lastFailureAt, now time.Time, cooldown time.Duration) bool {
	if consecutiveFailures < threshold {
		return true
	}
	return now.Sub(lastFailureAt) >= cooldown
}
package sample

import (
	"errors"
	"testing"
	"time"
)

func TestValidateEmail(t *testing.T) {
	if err := ValidateEmail("a@b.com"); err != nil {
		t.Errorf("ValidateEmail(valid) = %v, want nil", err)
	}
	cases := []string{"", "no-at-sign", "@b.com", "a@", "a@b", "a b@c.com"}
	for _, c := range cases {
		if err := ValidateEmail(c); err == nil {
			t.Errorf("ValidateEmail(%q) = nil, want error", c)
		}
	}
}

func TestMaskSensitive(t *testing.T) {
	if got := MaskSensitive("4111111111111111", 4); got != "************1111" {
		t.Errorf("MaskSensitive = %q", got)
	}
	if got := MaskSensitive("abc", 10); got != "abc" {
		t.Errorf("MaskSensitive(short) = %q, want unchanged", got)
	}
}

func TestDeduplicate(t *testing.T) {
	got := DeduplicateStrings([]string{"a", "b", "a", "c", "b"})
	want := []string{"a", "b", "c"}
	if len(got) != len(want) {
		t.Fatalf("Deduplicate = %v, want %v", got, want)
	}
	for i := range want {
		if got[i] != want[i] {
			t.Errorf("Deduplicate[%d] = %q, want %q", i, got[i], want[i])
		}
	}
}

func TestChunkSlice(t *testing.T) {
	chunks, err := ChunkSlice([]int{1, 2, 3, 4, 5}, 2)
	if err != nil {
		t.Fatalf("unexpected error: %v", err)
	}
	if len(chunks) != 3 || len(chunks[2]) != 1 {
		t.Errorf("ChunkSlice = %v", chunks)
	}
	if _, err := ChunkSlice([]int{1}, 0); err == nil {
		t.Error("ChunkSlice(size=0) err = nil, want error")
	}
}

func TestPaginate(t *testing.T) {
	items := []string{"a", "b", "c", "d", "e"}
	got, err := Paginate(items, 2, 2)
	if err != nil || len(got) != 2 || got[0] != "c" {
		t.Errorf("Paginate(page2) = %v, %v", got, err)
	}
	got, err = Paginate(items, 3, 2)
	if err != nil || len(got) != 1 || got[0] != "e" {
		t.Errorf("Paginate(last page) = %v, %v", got, err)
	}
	if _, err := Paginate(items, 0, 2); err == nil {
		t.Error("Paginate(page=0) err = nil, want error")
	}
}

func TestMergeConfigs(t *testing.T) {
	base := map[string]string{"a": "1", "b": "2"}
	override := map[string]string{"b": "override", "c": "3"}
	got := MergeConfigs(base, override)
	if got["a"] != "1" || got["b"] != "override" || got["c"] != "3" {
		t.Errorf("MergeConfigs = %v", got)
	}
	if base["b"] != "2" {
		t.Error("MergeConfigs mutated base")
	}
}

func TestTruncateWithEllipsis(t *testing.T) {
	if got := TruncateWithEllipsis("hello world", 8); got != "hello..." {
		t.Errorf("TruncateWithEllipsis = %q, want %q", got, "hello...")
	}
	if got := TruncateWithEllipsis("hi", 8); got != "hi" {
		t.Errorf("TruncateWithEllipsis(short) = %q, want unchanged", got)
	}
}

func TestValidatePasswordStrength(t *testing.T) {
	if got := ValidatePasswordStrength("Aa1!aaaa"); len(got) != 0 {
		t.Errorf("ValidatePasswordStrength(strong) = %v, want none", got)
	}
	got := ValidatePasswordStrength("weak")
	if len(got) == 0 {
		t.Error("ValidatePasswordStrength(weak) = none, want failures")
	}
}

func TestRetryWithBackoff(t *testing.T) {
	attempts := 0
	var slept []time.Duration
	sleep := func(d time.Duration) { slept = append(slept, d) }

	err := RetryWithBackoff(func() error {
		attempts++
		if attempts < 3 {
			return errors.New("not yet")
		}
		return nil
	}, 5, sleep)

	if err != nil {
		t.Fatalf("RetryWithBackoff = %v, want nil", err)
	}
	if attempts != 3 {
		t.Errorf("attempts = %d, want 3", attempts)
	}
	if len(slept) != 2 {
		t.Errorf("sleep called %d times, want 2", len(slept))
	}

	err = RetryWithBackoff(func() error { return errors.New("always fails") }, 2, sleep)
	if err == nil {
		t.Error("RetryWithBackoff(always fails) = nil, want error")
	}
}

func TestCircuitBreakerAllow(t *testing.T) {
	now := time.Now()
	if !CircuitBreakerAllow(1, 3, now.Add(-time.Minute), now, time.Minute) {
		t.Error("below threshold should allow")
	}
	if CircuitBreakerAllow(3, 3, now, now, time.Minute) {
		t.Error("just tripped, within cooldown should NOT allow")
	}
	if !CircuitBreakerAllow(3, 3, now.Add(-2*time.Minute), now, time.Minute) {
		t.Error("past cooldown should allow again")
	}
}
